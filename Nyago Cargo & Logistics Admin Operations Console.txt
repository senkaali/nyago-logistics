    try {
      const stored = localStorage.getItem('nyago_admin_shipments_v3');
      if (stored) {
        const parsed = JSON.parse(stored);
        if (Array.isArray(parsed) && parsed.length > 0) return parsed;
      }
    } catch (e) {
      console.warn(e);
    }
    return INITIAL_FALLBACK_SHIPMENTS;
  });

  const [isLiveConnected, setIsLiveConnected] = useState(false);
  const [selectedWaybillShipment, setSelectedWaybillShipment] = useState(null);
  const [toastMessage, setToastMessage] = useState(null);

  const showToast = useCallback((msg) => {
    setToastMessage(msg);
    setTimeout(() => setToastMessage(null), 3500);
  }, []);

  // Fetch from Supabase and listen for real-time changes
  const fetchShipments = useCallback(async () => {
    try {
      const { data, error } = await supabase
        .from('shipments')
        .select('*')
        .order('created_at', { ascending: false });

      if (!error && data && data.length > 0) {
        setShipments(data);
        localStorage.setItem('nyago_admin_shipments_v3', JSON.stringify(data));
      }
    } catch (err) {
      console.warn('Initial fetch using local cache:', err);
    }
  }, []);

  useEffect(() => {
    fetchShipments();

    // Supabase Realtime WebSocket subscription
    const channel = supabase
      .channel('nyago_admin_channel')
      .on(
        'postgres_changes',
        { event: '*', schema: 'public', table: 'shipments' },
        (payload) => {
          if (payload.eventType === 'INSERT') {
            setShipments((prev) => {
              const exists = prev.some(
                (s) => s.id === payload.new.id || s.tracking_id === payload.new.tracking_id
              );
              if (exists) return prev;
              const updated = [payload.new, ...prev];
              localStorage.setItem('nyago_admin_shipments_v3', JSON.stringify(updated));
              return updated;
            });
            showToast(`New client consignment received: ${payload.new.tracking_id}`);
          } else if (payload.eventType === 'UPDATE') {
            setShipments((prev) => {
              const updated = prev.map((item) =>
                item.id === payload.new.id || item.tracking_id === payload.new.tracking_id
                  ? payload.new
                  : item
              );
              localStorage.setItem('nyago_admin_shipments_v3', JSON.stringify(updated));
              return updated;
            });
            showToast(`Consignment ${payload.new.tracking_id} updated: ${payload.new.current_status}`);
          } else if (payload.eventType === 'DELETE') {
            setShipments((prev) => {
              const updated = prev.filter((item) => item.id !== payload.old.id);
              localStorage.setItem('nyago_admin_shipments_v3', JSON.stringify(updated));
              return updated;
            });
          }
        }
      )
      .subscribe((status) => {
        setIsLiveConnected(status === 'SUBSCRIBED');
      });

    return () => {
      supabase.removeChannel(channel);
    };
  }, [fetchShipments, showToast]);

  return (

          <Radio className="w-4 h-4 text-amber-400 animate-pulse" />

        </div>

      {/* Top Operations Header */}
      <header className="sticky top-0 z-40 bg-slate-900 border-b border-amber-500/30 shadow-lg">

                  NYAGO OPERATIONS

                  DISPATCH CONSOLE
                </span>
              </div>

                <span className={`w-2 h-2 rounded-full ${isLiveConnected ? 'bg-emerald-400 animate-pulse' : 'bg-amber-400'}`} />
                <span>Riyadh Central (RUH) &bull; Jeddah Port Dispatch &bull; {isLiveConnected ? 'Supabase Live' : 'Connecting...'}</span>

          {/* Desktop Navigation */}
          <nav className="hidden md:flex items-center gap-1 bg-slate-950/80 p-1.5 rounded-xl border border-slate-800">

// 3. ADMIN MANIFEST WORKSPACE VIEW
// ==========================================
function AdminManifestView({ shipments, onOpenWaybill, onOpenNewBooking, showToast }) {
  const [searchQuery, setSearchQuery] = useState('');
  const [statusFilter, setStatusFilter] = useState('ALL');
  const [modeFilter, setModeFilter] = useState('ALL');
  const [countryFilter, setCountryFilter] = useState('ALL');

  // Status edit modal state
  const [editingShipment, setEditingShipment] = useState(null);
  const [newStatus, setNewStatus] = useState('In Transit');
  const [customLocation, setCustomLocation] = useState('');
  const [customNotes, setCustomNotes] = useState('');
  const [carrierRef, setCarrierRef] = useState('');
  const [agentName, setAgentName] = useState('Operations Officer');
  const [isUpdating, setIsUpdating] = useState(false);
  const [actionError, setActionError] = useState(null);

  // KPIs
  const kpis = useMemo(() => {
    const totalCount = shipments.length;
    const inTransit = shipments.filter((s) => s.current_status === 'In Transit').length;
    const customs = shipments.filter(
      (s) =>
        s.current_status === 'Customs Cleared (KSA ZATCA)' ||
        s.current_status === 'Arrived at Destination Port/Airport' ||
        s.current_status === 'Customs Cleared (East Africa)'
    ).length;
    const delivered = shipments.filter((s) => s.current_status === 'Delivered').length;
    const pendingIntake = shipments.filter(
      (s) => s.current_status === 'Booking Created' || s.current_status === 'Received at KSA Hub'
    ).length;
    const totalRevenueSar = shipments.reduce((acc, s) => acc + (Number(s.total_cost_sar) || 0), 0);
    const totalWeightKg = shipments.reduce((acc, s) => acc + (Number(s.weight_kg) || 0), 0);
    const airVolumeKg = shipments
      .filter((s) => s.cargo_type === 'Air Freight')
      .reduce((acc, s) => acc + (Number(s.weight_kg) || 0), 0);
    const seaVolumeKg = shipments
      .filter((s) => s.cargo_type === 'Sea Freight')
      .reduce((acc, s) => acc + (Number(s.weight_kg) || 0), 0);

    return {
      totalCount,
      inTransit,
      customs,
      delivered,
      pendingIntake,
      totalRevenueSar,
      totalWeightKg,
      airVolumeKg,
      seaVolumeKg,
    };
  }, [shipments]);

  // Filtered shipments
  const filteredShipments = useMemo(() => {
    return shipments.filter((s) => {
      const q = searchQuery.toLowerCase();
      const matchesSearch =
        searchQuery === '' ||
        s.tracking_id?.toLowerCase().includes(q) ||
        s.sender_name?.toLowerCase().includes(q) ||
        s.receiver_name?.toLowerCase().includes(q) ||
        s.receiver_city?.toLowerCase().includes(q) ||
        s.sender_phone?.includes(searchQuery) ||
        s.receiver_phone?.includes(searchQuery);

      const matchesStatus = statusFilter === 'ALL' || s.current_status === statusFilter;
      const matchesMode = modeFilter === 'ALL' || s.cargo_type === modeFilter;
      const matchesCountry = countryFilter === 'ALL' || s.receiver_country === countryFilter;

      return matchesSearch && matchesStatus && matchesMode && matchesCountry;
    });
  }, [shipments, searchQuery, statusFilter, modeFilter, countryFilter]);

  // Update Shipment Status via Supabase
  const updateShipmentStatus = async (
    trackingId,
    targetStatus,
    location,
    notes,
    carrierIdentifier,
    officerName
  ) => {
    setIsUpdating(true);
    setActionError(null);
    try {
      const { data: currentShipment, error: fetchErr } = await supabase
        .from('shipments')
        .select('events')
        .eq('tracking_id', trackingId)
        .single();

      if (fetchErr) throw fetchErr;

      const newEvent = {
        id: `evt-${Date.now()}`,
        timestamp: new Date().toISOString(),
        status: targetStatus,
        location,
        notes,
        carrierRef: carrierIdentifier || undefined,
        updatedBy: officerName || 'Nyago Operations Desk',
      };

      const updatedEvents = [newEvent, ...(currentShipment?.events || [])];

      const { error: updateErr } = await supabase
        .from('shipments')
        .update({
          current_status: targetStatus,
          events: updatedEvents,
          updated_at: new Date().toISOString(),
        })
        .eq('tracking_id', trackingId);

      if (updateErr) throw updateErr;

      showToast(`Consignment ${trackingId} transitioned to ${targetStatus}`);
    } catch (err) {
      console.error('Failed to update status:', err);
      setActionError(err.message || 'Failed to update consignment status.');
    } finally {
      setIsUpdating(false);
    }
  };

  const handleOpenStatusModal = (shipment) => {
    setEditingShipment(shipment);

    const currentIndex = STATUS_STEPS.indexOf(shipment.current_status);
    if (currentIndex < STATUS_STEPS.length - 1) {
      const nextStep = STATUS_STEPS[currentIndex + 1];
      const defaultLoc =
        nextStep === 'Received at KSA Hub'
          ? `${shipment.sender_city} Logistics Hub`
          : nextStep === 'Customs Cleared (KSA ZATCA)'
          ? `${shipment.sender_city} International Port/Airport ZATCA`
          : nextStep === 'In Transit'
          ? shipment.cargo_type === 'Air Freight'
            ? 'Air Cargo Flight en route'
            : 'Maritime vessel en route via Red Sea'
          : nextStep === 'Arrived at Destination Port/Airport'
          ? shipment.cargo_type === 'Air Freight'
            ? 'Entebbe EBB / Nairobi NBO Airport'
            : 'Port of Mombasa Berth'
          : nextStep === 'Customs Cleared (East Africa)'
          ? `${shipment.receiver_country} Revenue Authority Customs`
          : nextStep === 'Out for Delivery / Ready for Pickup'
          ? `Nyago Express Depot (${shipment.receiver_city})`
          : `Delivered to ${shipment.receiver_name}`;

      const defaultCarrier =
        shipment.cargo_type === 'Air Freight' ? 'Direct Cargo Express' : 'Sea Container Consolidation';

      await updateShipmentStatus(
        shipment.tracking_id,
        nextStep,
        defaultLoc,
        `Consignment advanced to ${nextStep}.`,
        defaultCarrier,
        'Nyago Admin Desk'
      );
    }
  };

    e.preventDefault();

      editingShipment.tracking_id,

  };

      s.package_count,
      `"${(s.goods_description || '').replace(/"/g, '""')}"`,
      s.total_cost_sar,
      s.payment_status,
      `"${s.current_status || ''}"`,
      s.created_at,
    ]);

    const csvContent =
      'data:text/csv;charset=utf-8,' +
      [headers.join(','), ...rows.map((e) => e.join(','))].join('\n');
    const encodedUri = encodeURI(csvContent);
    const link = document.createElement('a');
    link.setAttribute('href', encodedUri);
    link.setAttribute('download', `nyago_cargo_manifest_${new Date().toISOString().split('T')[0]}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

      {/* Action Bar */}
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 pb-4 border-b border-slate-800">
        <div>
          <h2 className="text-2xl font-bold text-white tracking-tight">Consignment Manifest Control</h2>

            Real-time tracking transition desk &bull; Changes reflect immediately on customer tracker
          </p>
        </div>
        <div className="flex items-center gap-3">
          <button
            type="button"
            onClick={handleExportCSV}
            className="flex items-center gap-2 bg-slate-900 hover:bg-slate-800 text-slate-300 hover:text-white px-3.5 py-2 rounded-xl text-xs font-semibold border border-slate-800 transition-colors"

              </div>
              <button
                type="button"
                onClick={() => setEditingShipment(null)}
                className="text-slate-400 hover:text-white p-1 rounded-lg"

                <X className="w-5 h-5" />

            </div>

            <form onSubmit={handleSaveStatusUpdate} className="space-y-4 text-xs">

                <label className="block text-slate-300 font-medium mb-1.5">New Milestone Status</label>

                  onChange={(e) => setNewStatus(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-sm text-white focus:outline-none focus:border-amber-500"
                >
                  {STATUS_STEPS.map((step) => (
                    <option key={step} value={step}>

              <div>
                <label className="block text-slate-300 font-medium mb-1.5">Current Physical Location / Hub</label>
                <input
                  type="text"
                  required
                  value={customLocation}
                  onChange={(e) => setCustomLocation(e.target.value)}
                  placeholder="e.g. Entebbe Airport Cargo Bay or Kampala Central Hub"
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-sm text-white focus:outline-none focus:border-amber-500"
                />
              </div>

              <div>
                <label className="block text-slate-300 font-medium mb-1.5">Carrier / Flight / Vessel Identifier</label>
                <input
                  type="text"
                  value={carrierRef}
                  onChange={(e) => setCarrierRef(e.target.value)}
                  placeholder="e.g. Flight SV-844 or Container MSC Aurelia Voy 24B"
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-sm text-white focus:outline-none focus:border-amber-500 font-mono"
                />
              </div>

              <div>
                <label className="block text-slate-300 font-medium mb-1.5">Log / Audit Notes</label>
                <textarea
                  rows={2}
                  required
                  value={customNotes}
                  onChange={(e) => setCustomNotes(e.target.value)}
                  placeholder="Describe checkpoint progress, customs release, or inspection notes..."
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-sm text-white focus:outline-none focus:border-amber-500"
                />
              </div>

              <div>
                <label className="block text-slate-300 font-medium mb-1.5">Updating Officer Name</label>
                <input
                  type="text"
                  required
                  value={agentName}
                  onChange={(e) => setAgentName(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-sm text-white focus:outline-none focus:border-amber-500"
                />
              </div>

              <div className="pt-3 border-t border-slate-800 flex items-center justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setEditingShipment(null)}
                  className="px-4 py-2 text-slate-400 hover:text-white rounded-xl"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  disabled={isUpdating}
                  className="px-5 py-2.5 bg-amber-500 hover:bg-amber-400 disabled:opacity-50 text-slate-950 font-bold rounded-xl transition-all"
                >
                  {isUpdating ? 'Saving...' : 'Save & Broadcast Update'}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}

function AdminIntakeBookingView({ onBookingSuccess }) {
  const [senderName, setSenderName] = useState('');
  const [senderPhone, setSenderPhone] = useState('+966 ');
  const [senderCity, setSenderCity] = useState('Riyadh');
  const [senderIqama, setSenderIqama] = useState('');

  const [receiverName, setReceiverName] = useState('');
  const [receiverPhone, setReceiverPhone] = useState('+256 ');
  const [receiverCountry, setReceiverCountry] = useState('Uganda');
  const [receiverCity, setReceiverCity] = useState('Kampala');
  const [receiverAddress, setReceiverAddress] = useState('Central Depot (Collection)');

  const [cargoType, setCargoType] = useState('Air Freight');
  const [weightKg, setWeightKg] = useState(15);
  const [packageType, setPackageType] = useState('Carton Box');
  const [packageCount, setPackageCount] = useState(1);
  const [goodsDescription, setGoodsDescription] = useState('');
  const [declaredValueSar, setDeclaredValueSar] = useState(500);
  const [includeInsurance, setIncludeInsurance] = useState(true);
  const [paymentStatus, setPaymentStatus] = useState('Paid');
  const [paymentMethod, setPaymentMethod] = useState('Cash at Depot');

  const [isSubmitting, setIsSubmitting] = useState(false);
  const [errorMsg, setErrorMsg] = useState(null);

  const calculations = useMemo(() => {
    const ratePerKg = cargoType === 'Air Freight' ? FREIGHT_RATES.airFreightPerKg : FREIGHT_RATES.seaFreightPerKg;
    const safeWeight = Math.max(0.5, Number(weightKg) || 0);
    const baseCost = Math.round(safeWeight * ratePerKg * 100) / 100;
    const handling = FREIGHT_RATES.handlingFee;
    const insuranceFee = includeInsurance
      ? Math.round((Math.max(0, Number(declaredValueSar) || 0) * FREIGHT_RATES.insuranceRate) * 100) / 100
      : 0;
    const totalSar = Math.round((baseCost + handling + insuranceFee) * 100) / 100;

    return {
      ratePerKg,
      baseCost,
      handling,
      insuranceFee,
      totalSar,
    };
  }, [cargoType, weightKg, includeInsurance, declaredValueSar]);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setIsSubmitting(true);
    setErrorMsg(null);

    const trackingId = generateTrackingCode();
    const now = new Date();
    const deliveryDays = cargoType === 'Air Freight' ? 4 : 28;
    const estDate = new Date();
    estDate.setDate(now.getDate() + deliveryDays);

    const initialEvent = {
      id: `evt-${Date.now()}-1`,
      timestamp: now.toISOString(),
      status: 'Booking Created',
      location: `${senderCity} Nyago Freight Depot`,
      notes: `Walk-in registration at counter. ${packageCount} package(s), ${weightKg} kg (${cargoType}).`,
      updatedBy: 'Operations Desk Officer',
    };

    const payload = {
      tracking_id: trackingId,
      cargo_type: cargoType,
      sender_name: senderName.trim(),
      sender_phone: senderPhone.trim(),
      sender_city: senderCity,
      sender_id_iqama: senderIqama.trim() || null,
      receiver_name: receiverName.trim(),
      receiver_phone: receiverPhone.trim(),
      receiver_country: receiverCountry,
      receiver_city: receiverCity.trim(),
      receiver_address: receiverAddress.trim(),
      package_type: packageType,
      package_count: Number(packageCount) || 1,
      weight_kg: Number(weightKg),

        ...payload,
        id: `ship-${Date.now()}`,

// 5. ADMIN ANALYTICS WORKSPACE VIEW
// ==========================================
function AdminAnalyticsView({ shipments }) {
  const stats = useMemo(() => {
    const totalRevenue = shipments.reduce((sum, s) => sum + (Number(s.total_cost_sar) || 0), 0);
    const airRevenue = shipments
      .filter((s) => s.cargo_type === 'Air Freight')
      .reduce((sum, s) => sum + (Number(s.total_cost_sar) || 0), 0);
    const seaRevenue = shipments
      .filter((s) => s.cargo_type === 'Sea Freight')
      .reduce((sum, s) => sum + (Number(s.total_cost_sar) || 0), 0);
    const totalWeight = shipments.reduce((sum, s) => sum + (Number(s.weight_kg) || 0), 0);
    const airWeight = shipments
      .filter((s) => s.cargo_type === 'Air Freight')
      .reduce((sum, s) => sum + (Number(s.weight_kg) || 0), 0);
    const seaWeight = shipments
      .filter((s) => s.cargo_type === 'Sea Freight')
      .reduce((sum, s) => sum + (Number(s.weight_kg) || 0), 0);

    const ugandaCount = shipments.filter((s) => s.receiver_country === 'Uganda').length;
    const kenyaCount = shipments.filter((s) => s.receiver_country === 'Kenya').length;
    const otherCount = shipments.length - ugandaCount - kenyaCount;

    return {
      totalRevenue,
      airRevenue,
      seaRevenue,
      totalWeight,
      airWeight,
      seaWeight,
      ugandaCount,
      kenyaCount,
      otherCount,
      totalCount: shipments.length,
    };
  }, [shipments]);

  return (
    <div className="space-y-8">
      <div className="border-b border-slate-800 pb-4">
        <h2 className="text-2xl font-bold text-white tracking-tight">Logistics Operations Analytics</h2>
        <p className="text-xs text-slate-400 mt-1">
          Financial throughput, payload tonnage, and corridor distribution across East Africa.
        </p>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 space-y-2">
          <div className="text-xs text-slate-400 flex items-center justify-between">
            <span>Gross Revenue</span>
            <TrendingUp className="w-4 h-4 text-emerald-400" />
          </div>
          <div className="text-3xl font-extrabold text-white font-mono tabular-nums">
            {stats.totalRevenue.toLocaleString('en-US', { minimumFractionDigits: 2 })} SAR
          </div>
          <div className="text-xs text-slate-500 font-mono">100% of collected logistics fees</div>
        </div>

        <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 space-y-2">
          <div className="text-xs text-slate-400 flex items-center justify-between">
            <span>Air Freight (23 SAR/kg)</span>
            <Plane className="w-4 h-4 text-amber-400" />
          </div>
          <div className="text-3xl font-extrabold text-amber-400 font-mono tabular-nums">
            {stats.airRevenue.toLocaleString('en-US', { minimumFractionDigits: 2 })} SAR
          </div>
          <div className="text-xs text-slate-500 font-mono">
            {stats.totalRevenue > 0
              ? `${Math.round((stats.airRevenue / stats.totalRevenue) * 100)}% of total revenue`
              : '0%'}
          </div>
        </div>

        <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 space-y-2">
          <div className="text-xs text-slate-400 flex items-center justify-between">
            <span>Sea Freight (13 SAR/kg)</span>
            <Ship className="w-4 h-4 text-sky-400" />
          </div>
          <div className="text-3xl font-extrabold text-sky-400 font-mono tabular-nums">

          <div className="grid grid-cols-1 sm:grid-cols-2 border border-slate-950 divide-y sm:divide-y-0 sm:divide-x divide-slate-950">
            <div className="p-3.5 space-y-1">
              <div className="text-[10px] font-bold uppercase text-slate-500 tracking-wider">
                1. Shipper / Consignor (Saudi Arabia)
              </div>
              <div className="font-bold text-sm text-slate-950">{shipment.sender_name}</div>
              <div className="font-mono text-slate-700">{shipment.sender_phone}</div>
              <div className="text-slate-700">Origin Depot: {shipment.sender_city}, KSA</div>
              {shipment.sender_id_iqama && (
                <div className="font-mono text-[10px] text-slate-600">
                  National ID / Iqama: {shipment.sender_id_iqama}
                </div>
              )}
            </div>

            <div className="p-3.5 space-y-1">
              <div className="text-[10px] font-bold uppercase text-slate-500 tracking-wider">
                2. Consignee / Recipient ({shipment.receiver_country})
              </div>
              <div className="font-bold text-sm text-slate-950">{shipment.receiver_name}</div>
              <div className="font-mono text-slate-700">{shipment.receiver_phone}</div>
              <div className="text-slate-800 font-medium">
                {shipment.receiver_address}, {shipment.receiver_city}
              </div>
              <div className="font-bold text-slate-950">{shipment.receiver_country}</div>
            </div>
          </div>

          <div className="grid grid-cols-4 border border-slate-950 divide-x divide-slate-950 text-center text-[11px]">
            <div className="p-2.5">
              <span className="text-[10px] text-slate-500 block uppercase font-bold">Departure Port</span>
              <span className="font-bold text-slate-950">
                {shipment.sender_city === 'Jeddah'
                  ? 'JED / Jeddah Port'
                  : shipment.sender_city === 'Riyadh'
                  ? 'RUH / King Khalid Intl'
                  : 'DMM / King Fahd Intl'}
              </span>
            </div>
            <div className="p-2.5">
              <span className="text-[10px] text-slate-500 block uppercase font-bold">Destination Hub</span>
              <span className="font-bold text-slate-950">
                {shipment.receiver_country === 'Uganda'
                  ? 'EBB / Kampala Station'
                  : shipment.receiver_country === 'Kenya'
                  ? 'NBO / Port of Mombasa'
                  : `${shipment.receiver_city} Hub`}
              </span>
            </div>
            <div className="p-2.5">
              <span className="text-[10px] text-slate-500 block uppercase font-bold">Service Mode</span>
              <span className="font-bold text-slate-950">{shipment.cargo_type}</span>
            </div>
            <div className="p-2.5">
              <span className="text-[10px] text-slate-500 block uppercase font-bold">Target Delivery</span>
              <span className="font-mono font-bold text-slate-950">{shipment.estimated_delivery}</span>
            </div>
          </div>

          <div className="border border-slate-950">
            <table className="w-full text-left text-[11px]">
              <thead className="bg-slate-100 border-b border-slate-950 text-[10px] uppercase font-bold text-slate-700">
                <tr>
                  <th className="p-2 border-r border-slate-950">Pieces</th>
                  <th className="p-2 border-r border-slate-950">Package Type</th>
                  <th className="p-2 border-r border-slate-950">Gross Weight</th>
                  <th className="p-2 border-r border-slate-950">Declared Contents</th>
                  <th className="p-2 text-right">Declared Value</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-300">
                <tr>
                  <td className="p-2 border-r border-slate-950 font-mono font-bold text-center">
                    {shipment.package_count}
                  </td>
                  <td className="p-2 border-r border-slate-950 font-medium">{shipment.package_type}</td>
                  <td className="p-2 border-r border-slate-950 font-mono font-bold">
                    {shipment.weight_kg} kg
                  </td>
                  <td className="p-2 border-r border-slate-950 font-medium">
                    {shipment.goods_description}
                  </td>
                  <td className="p-2 text-right font-mono font-bold">
                    {shipment.declared_value_sar || 0} SAR
                  </td>
                </tr>
              </tbody>
            </table>
          </div>

          <div className="grid grid-cols-2 gap-4">
            <div className="border border-slate-950 p-3 text-[11px] space-y-1">
              <div className="font-bold uppercase text-[10px] text-slate-600">Handling &amp; Customs</div>
              <div className="flex items-center gap-1.5 text-emerald-700 font-semibold">
                <ShieldCheck className="w-3.5 h-3.5" />
                Security Screened &bull; Customs Export Bonded
              </div>
              <p className="text-[10px] text-slate-600 leading-tight">
                Consignment verified safe for air transport or ocean container stowage under KSA ZATCA &amp; IATA/IMO regulations.
              </p>
            </div>
            <div className="border border-slate-950 p-3 text-[11px] space-y-1 text-right">
              <div className="flex justify-between text-slate-600">
                <span>Freight ({shipment.rate_per_kg} SAR/kg):</span>
                <span className="font-mono">{Number(shipment.base_cost_sar || 0).toFixed(2)} SAR</span>
              </div>
              <div className="flex justify-between text-slate-600">
                <span>Documentation / Intake:</span>
                <span className="font-mono">{Number(shipment.handling_fee_sar || 0).toFixed(2)} SAR</span>
              </div>
              {shipment.insurance_fee_sar > 0 && (
                <div className="flex justify-between text-slate-600">
                  <span>Transit Insurance:</span>
                  <span className="font-mono">{Number(shipment.insurance_fee_sar || 0).toFixed(2)} SAR</span>
                </div>
              )}
              <div className="flex justify-between font-bold text-sm text-slate-950 pt-1 border-t border-slate-300">
                <span>Total Amount:</span>
                <span className="font-mono">{Number(shipment.total_cost_sar || 0).toFixed(2)} SAR</span>
              </div>
              <div className="text-[10px] font-mono text-slate-600">
                Status: <strong className="text-slate-950 uppercase">{shipment.payment_status}</strong>
              </div>
            </div>
          </div>

          <div className="pt-4 border-t-2 border-slate-950 flex flex-col sm:flex-row items-center justify-between gap-4">
            <div className="space-y-1 text-center sm:text-left">
              <div className="inline-flex items-center gap-1 h-12 px-2 bg-white">
                {Array.from({ length: 48 }).map((_, i) => (
                  <div
                    key={i}
                    className={`bg-slate-950 h-full ${
                      i % 7 === 0 || i % 5 === 0 ? 'w-1' : i % 3 === 0 ? 'w-1.5' : 'w-0.5'
                    } ${i % 11 === 0 ? 'opacity-0' : 'opacity-100'}`}
                  />
                ))}
              </div>
              <div className="font-mono text-center font-bold tracking-widest text-xs text-slate-950">
                *{shipment.tracking_id}*
              </div>
            </div>

            <div className="grid grid-cols-2 gap-4 text-[10px] text-center">
              <div className="w-32 border-b border-slate-950 pb-1">
                <div className="h-8" />
                <span>Shipper Signature</span>
              </div>
              <div className="w-32 border-b border-slate-950 pb-1">
                <div className="h-8 font-mono text-slate-400 flex items-center justify-center">
                  [STAMP: NYAGO]
                </div>
                <span>Authorized Cargo Agent</span>
